TOL = 1e-12


def kron3(A: np.ndarray, B: np.ndarray, C: np.ndarray) -> np.ndarray:
  return np.kron(np.kron(A, B), C)


def spectrum_counts(mat: np.ndarray, tol: float = 1e-8) -> dict[int, int]:
  """Group real Hermitian eigenvalues by nearest integer within tolerance."""
  ev = np.linalg.eigvalsh(mat)
  rounded = np.rint(ev).astype(int)
  assert np.max(np.abs(ev - rounded)) < tol, "Non-integer eigenvalue detected"
  vals, counts = np.unique(rounded, return_counts=True)
  return dict(zip(vals.tolist(), counts.tolist()))


def vn_entropy(rho: np.ndarray) -> float:
  ev = np.linalg.eigvalsh(rho)
  ev = ev[ev > 1e-15]
  return float(-np.sum(ev * np.log(ev)))
